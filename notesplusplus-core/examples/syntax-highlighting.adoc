= Syntax Highlighting Demo
:toc:

Code blocks with a `[source,language]` annotation get automatic syntax coloring.
The highlighter recognizes common language names and their short aliases.

== Rust

[source,rust]
----
fn fibonacci(n: u32) -> u32 {
    match n {
        0 => 0,
        1 => 1,
        _ => fibonacci(n - 1) + fibonacci(n - 2),
    }
}

fn main() {
    for i in 0..10 {
        println!("fib({}) = {}", i, fibonacci(i));
    }
}
----

== Python

Aliases: `py`, `python`

[source,python]
----
import asyncio

async def fetch_data(url: str) -> dict:
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as resp:
            return await resp.json()

async def main():
    data = await fetch_data("https://api.example.com/items")
    for item in data["results"]:
        print(f"{item['name']}: {item['value']:.2f}")

asyncio.run(main())

== JavaScript / TypeScript

Aliases: `js`, `javascript`, `ts`, `typescript`

[source,javascript]
----
const debounce = (fn, delay) => {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
};

const search = debounce(async (query) => {
  const res = await fetch(`/api/search?q=${encodeURIComponent(query)}`);
  const { results } = await res.json();
  renderResults(results);
}, 300);

== SQL

[source,sql]
----
SELECT
    n.title,
    n.created_at,
    GROUP_CONCAT(t.name, ', ') AS tags,
    snippet(notes_fts, 0, '<mark>', '</mark>', '...', 32) AS match_preview
FROM notes_fts
JOIN notes n ON n.id = notes_fts.rowid
LEFT JOIN note_tags t ON t.note_id = n.id
WHERE notes_fts MATCH 'syntax NEAR highlighting'
GROUP BY n.id
ORDER BY rank
LIMIT 20;
----

== Go

[source,go]
----
package main

import (
    "fmt"
    "net/http"
    "sync"
)

var (
    mu    sync.RWMutex
    cache = make(map[string]string)
)

func handler(w http.ResponseWriter, r *http.Request) {
    mu.RLock()
    val, ok := cache[r.URL.Path]
    mu.RUnlock()

    if !ok {
        http.Error(w, "not found", http.StatusNotFound)
        return
    }
    fmt.Fprint(w, val)
}

func main() {
    http.HandleFunc("/", handler)
    fmt.Println("Listening on :8080")
    http.ListenAndServe(":8080", nil)
}
----

== C

[source,c]
----
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    char *key;
    char *value;
    struct Node *next;
} Node;

Node *node_create(const char *key, const char *value) {
    Node *n = malloc(sizeof(Node));
    n->key = strdup(key);
    n->value = strdup(value);
    n->next = NULL;
    return n;
}
----

== Without Language

Code blocks without a language annotation render as plain monospace text:

----
This is a plain code block.
No syntax coloring is applied.
Useful for output logs or pseudo-code.
----

== Language Aliases

The highlighter accepts both full names and short aliases:

[cols="1,1"]
|===
| Alias | Resolves to

| `rs`       | Rust
| `py`       | Python
| `js`       | JavaScript
| `ts`       | TypeScript
| `sh`       | Bash
| `rb`       | Ruby
| `yml`      | YAML
| `md`       | Markdown
| `cpp`      | C++
| `kt`       | Kotlin
| `toml`     | TOML
|===
